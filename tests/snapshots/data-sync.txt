Sync candles from Binance into SQLite

Usage: quantforge data sync [OPTIONS] --symbol <SYMBOL>

Options:
      --db <DB>
          SQLite database path [env: QF_DB=] [default: data/market.sqlite]
      --symbol <SYMBOL>
          
      --binance-base-url <BINANCE_BASE_URL>
          Binance API base URL. Use https://testnet.binance.vision/ for Spot testnet [env: QF_BINANCE_BASE_URL=] [default: https://api.binance.com/]
      --interval <INTERVAL>
          [default: 1m]
      --log-level <LOG_LEVEL>
          Log filter (trace, debug, info, warn, error or a full tracing filter expression) [env: QF_LOG=] [default: info]
      --start <START>
          RFC3339 start time. Omit to begin syncing from the current time
      --end <END>
          RFC3339 end time. Omit to keep syncing indefinitely
      --limit <LIMIT>
          [default: 1000]
      --follow
          When --end is set, keep polling until that end boundary is reached
      --poll-secs <POLL_SECS>
          [default: 5]
      --max-loops <MAX_LOOPS>
          
  -h, --help
          Print help
